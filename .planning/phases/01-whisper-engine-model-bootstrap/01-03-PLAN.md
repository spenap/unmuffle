```kotlin
package com.simonpena.transchat

import android.content.Context
import android.util.Log
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.security.MessageDigest

private const val TAG = "ModelManager"

/**
 * Manages the Whisper model file stored in app private storage.
 *
 * Model location: context.filesDir/models/ggml-small-q5_1.bin
 * Never stored on external storage — this is private, never backed up or shared.
 *
 * All heavy operations (SHA256 of 190MB file) must be called from a coroutine.
 */
object ModelManager {

    const val MODEL_FILENAME = "ggml-small-q5_1.bin"
    const val MODEL_URL =
        "https://huggingface.co/ggerganov/whisper.cpp/resolve/main/ggml-small-q5_1.bin"

    // SHA256 retrieved from HuggingFace 2026-02-27 — verify matches downloaded file
    const val MODEL_SHA256 =
        "ae85e4a935d7a567bd102fe55afc16bb595bdb618e11b2fc7591bc08120411bb"

    const val MODEL_SIZE_BYTES = 199_819_264L  // ~190 MB

    /** Absolute path to the model file in private storage. */
    fun modelFile(context: Context): File {
        val modelsDir = File(context.filesDir, "models")
        modelsDir.mkdirs()  // creates directory if it does not exist; filesDir always exists
        return File(modelsDir, MODEL_FILENAME)
    }

    /**
     * Returns true if the model file exists at the expected path.
     * Does NOT verify the SHA256 — call [isModelReady] for full verification.
     */
    fun modelExists(context: Context): Boolean = modelFile(context).exists()

    /**
     * Full model readiness check: file exists AND SHA256 matches.
     * Runs the SHA256 digest on a 190MB file — call from a coroutine on Dispatchers.IO.
     *
     * @return true if the model is present and has the correct checksum
     */
    suspend fun isModelReady(context: Context): Boolean = withContext(Dispatchers.IO) {
        val file = modelFile(context)
        if (!file.exists()) return@withContext false
        val hash = computeSha256(file)
        val match = hash == MODEL_SHA256
        if (!match) {
            Log.w(TAG, "SHA256 mismatch — expected $MODEL_SHA256, got $hash")
        }
        match
    }

    /**
     * Quick existence-only check (no SHA256). Use in MainActivity routing where
     * we want low latency — the full verify runs after download completes.
     */
    fun isModelPresentFast(context: Context): Boolean = modelFile(context).let {
        it.exists() && it.length() > 0L
    }

    /**
     * Delete the model file (e.g. on failed or corrupt download before retry).
     * Safe to call even if the file does not exist.
     */
    fun deleteModel(context: Context) {
        val file = modelFile(context)
        if (file.exists()) {
            val deleted = file.delete()
            Log.i(TAG, "Model file deleted: $deleted")
        }
    }

    /** Compute SHA256 hex digest of [file]. Reads in 8KB chunks. */
    suspend fun computeSha256(file: File): String = withContext(Dispatchers.IO) {
        val md = MessageDigest.getInstance("SHA-256")
        file.inputStream().buffered(8192).use { input ->
            val buffer = ByteArray(8192)
            var bytesRead: Int
            while (input.read(buffer).also { bytesRead = it } != -1) {
                md.update(buffer, 0, bytesRead)
            }
        }
        md.digest().joinToString("") { "%02x".format(it) }
    }
}
```

```kotlin
package com.simonpena.transchat

import android.app.Application
import android.app.DownloadManager
import android.content.Context
import android.net.ConnectivityManager
import android.net.NetworkCapabilities
import android.net.Uri
import android.util.Log
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.launch
import java.io.File

private const val TAG = "DownloadViewModel"

class DownloadViewModel(app: Application) : AndroidViewModel(app) {

    sealed class State {
        object Idle : State()
        data class Downloading(
            val progressFraction: Float,   // 0.0 - 1.0
            val bytesDownloaded: Long,
            val totalBytes: Long
        ) : State()
        object Verifying : State()
        object Complete : State()
        data class Failed(val reason: String) : State()
    }

    private val _state = MutableStateFlow<State>(State.Idle)
    val state: StateFlow<State> = _state

    /** True if device is on a cellular-only connection (no WiFi). */
    fun isOnCellularOnly(): Boolean {
        val cm = getApplication<Application>()
            .getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val caps = cm.getNetworkCapabilities(cm.activeNetwork) ?: return false
        return caps.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR) &&
                !caps.hasTransport(NetworkCapabilities.TRANSPORT_WIFI)
    }

    /**
     * Enqueue the model download via Android DownloadManager.
     *
     * @param allowCellular Set to true if user confirmed the cellular warning dialog.
     */
    fun startDownload(allowCellular: Boolean = false) {
        val context = getApplication<Application>()
        val destFile = ModelManager.modelFile(context)

        // Delete any partial file from a previous failed attempt
        if (destFile.exists()) destFile.delete()

        val downloadManager =
            context.getSystemService(Context.DOWNLOAD_SERVICE) as DownloadManager

        val networkTypes = if (allowCellular) {
            DownloadManager.Request.NETWORK_WIFI or DownloadManager.Request.NETWORK_MOBILE
        } else {
            DownloadManager.Request.NETWORK_WIFI
        }

        val request = DownloadManager.Request(Uri.parse(ModelManager.MODEL_URL))
            .setTitle("TransChat — Whisper Model")
            .setDescription("Downloading transcription model (190 MB)...")
            .setDestinationUri(Uri.fromFile(destFile))
            .setAllowedNetworkTypes(networkTypes)
            .setNotificationVisibility(
                DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED
            )

        val downloadId = downloadManager.enqueue(request)
        Log.i(TAG, "Download enqueued with id=$downloadId to ${destFile.absolutePath}")

        // Poll DownloadManager every 500ms — there is no callback API
        viewModelScope.launch {
            pollDownloadProgress(downloadManager, downloadId, destFile)
        }
    }

    /** Retry download after failure — resets state to Idle then starts again. */
    fun retryDownload(allowCellular: Boolean = false) {
        val context = getApplication<Application>()
        ModelManager.deleteModel(context)
        _state.value = State.Idle
        startDownload(allowCellular)
    }

    private suspend fun pollDownloadProgress(
        downloadManager: DownloadManager,
        downloadId: Long,
        destFile: File
    ) {
        while (true) {
            val query = DownloadManager.Query().setFilterById(downloadId)
            val cursor = downloadManager.query(query)
            cursor.use { c ->
                if (c.moveToFirst()) {
                    val status = c.getInt(c.getColumnIndexOrThrow(DownloadManager.COLUMN_STATUS))
                    val bytesDownloaded = c.getLong(
                        c.getColumnIndexOrThrow(DownloadManager.COLUMN_BYTES_DOWNLOADED_SO_FAR)
                    )
                    val totalBytes = c.getLong(
                        c.getColumnIndexOrThrow(DownloadManager.COLUMN_TOTAL_SIZE_BYTES)
                    )

                    when (status) {
                        DownloadManager.STATUS_RUNNING, DownloadManager.STATUS_PENDING -> {
                            val fraction = if (totalBytes > 0) {
                                bytesDownloaded.toFloat() / totalBytes.toFloat()
                            } else 0f
                            _state.value = State.Downloading(fraction, bytesDownloaded, totalBytes)
                        }
                        DownloadManager.STATUS_SUCCESSFUL -> {
                            Log.i(TAG, "Download complete — verifying SHA256")
                            _state.value = State.Verifying
                            verifyAndComplete(destFile)
                            return
                        }
                        DownloadManager.STATUS_FAILED -> {
                            val reason = c.getInt(
                                c.getColumnIndexOrThrow(DownloadManager.COLUMN_REASON)
                            )
                            Log.e(TAG, "Download failed with reason=$reason")
                            ModelManager.deleteModel(getApplication())
                            _state.value = State.Failed("Download failed (code $reason). Tap Retry.")
                            return
                        }
                    }
                }
            }
            delay(500)
        }
    }

    private suspend fun verifyAndComplete(destFile: File) {
        val context = getApplication<Application>()
        val isValid = ModelManager.isModelReady(context)
        if (isValid) {
            Log.i(TAG, "SHA256 verified — model ready")
            _state.value = State.Complete
        } else {
            Log.e(TAG, "SHA256 mismatch — deleting corrupt model")
            ModelManager.deleteModel(context)
            _state.value = State.Failed("Download corrupt. SHA256 mismatch. Tap Retry.")
        }
    }
}
```

```kotlin
package com.simonpena.transchat

import android.content.Intent
import android.os.Bundle
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import kotlinx.coroutines.launch

/**
 * Launcher Activity — transparent router. Never shown to user directly.
 * Checks model presence and routes to DownloadActivity or ReadyActivity.
 *
 * Uses isModelPresentFast (no SHA256) for quick routing. Full SHA256 check
 * happens after download in DownloadViewModel. This avoids a 2-3s delay on
 * every app launch just to verify a 190MB file.
 */
class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        // No setContentView — this Activity is purely a router
        routeToCorrectScreen()
    }

    private fun routeToCorrectScreen() {
        val destination = if (ModelManager.isModelPresentFast(this)) {
            Intent(this, ReadyActivity::class.java)
        } else {
            Intent(this, DownloadActivity::class.java)
        }
        startActivity(destination)
        finish()  // Remove MainActivity from back stack
    }
}
```

```kotlin
package com.simonpena.transchat

import android.content.Intent
import android.os.Bundle
import android.view.View
import android.widget.Button
import android.widget.ProgressBar
import android.widget.TextView
import androidx.activity.viewModels
import androidx.appcompat.app.AlertDialog
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import kotlinx.coroutines.launch

class DownloadActivity : AppCompatActivity() {

    private val viewModel: DownloadViewModel by viewModels()

    private lateinit var tvExplainer: TextView
    private lateinit var progressBar: ProgressBar
    private lateinit var tvProgress: TextView
    private lateinit var btnDownload: Button
    private lateinit var tvStatus: TextView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_download)

        tvExplainer = findViewById(R.id.tv_explainer)
        progressBar = findViewById(R.id.progress_bar)
        tvProgress = findViewById(R.id.tv_progress)
        btnDownload = findViewById(R.id.btn_download)
        tvStatus = findViewById(R.id.tv_status)

        btnDownload.setOnClickListener { onDownloadClicked() }

        lifecycleScope.launch {
            viewModel.state.collect { state -> renderState(state) }
        }
    }

    private fun onDownloadClicked() {
        if (viewModel.isOnCellularOnly()) {
            AlertDialog.Builder(this)
                .setTitle("Using Cellular Data")
                .setMessage("The transcription model is ~190 MB. Download on cellular data?")
                .setPositiveButton("Download anyway") { _, _ -> viewModel.startDownload(allowCellular = true) }
                .setNegativeButton("Wait for WiFi") { _, _ -> }
                .show()
        } else {
            viewModel.startDownload(allowCellular = false)
        }
    }

    private fun renderState(state: DownloadViewModel.State) {
        when (state) {
            is DownloadViewModel.State.Idle -> {
                btnDownload.isEnabled = true
                btnDownload.text = "Download (190 MB)"
                progressBar.visibility = View.GONE
                tvProgress.visibility = View.GONE
                tvStatus.visibility = View.GONE
            }
            is DownloadViewModel.State.Downloading -> {
                btnDownload.isEnabled = false
                btnDownload.text = "Downloading…"
                progressBar.visibility = View.VISIBLE
                progressBar.max = 100
                progressBar.progress = (state.progressFraction * 100).toInt()
                tvProgress.visibility = View.VISIBLE
                tvProgress.text = "${formatBytes(state.bytesDownloaded)} of ${formatBytes(state.totalBytes)}"
                tvStatus.visibility = View.GONE
            }
            is DownloadViewModel.State.Verifying -> {
                btnDownload.isEnabled = false
                btnDownload.text = "Verifying…"
                progressBar.progress = 100
                tvProgress.text = "Verifying download integrity…"
                tvStatus.visibility = View.GONE
            }
            is DownloadViewModel.State.Complete -> {
                startActivity(Intent(this, ReadyActivity::class.java))
                finish()
            }
            is DownloadViewModel.State.Failed -> {
                btnDownload.isEnabled = true
                btnDownload.text = "Retry"
                progressBar.visibility = View.GONE
                tvProgress.visibility = View.GONE
                tvStatus.visibility = View.VISIBLE
                tvStatus.text = state.reason
                btnDownload.setOnClickListener { viewModel.retryDownload() }
            }
        }
    }

    private fun formatBytes(bytes: Long): String {
        if (bytes <= 0) return "0 MB"
        val mb = bytes / (1024.0 * 1024.0)
        return "%.1f MB".format(mb)
    }
}
```
