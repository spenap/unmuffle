**Step 2: Create app/src/main/cpp/CMakeLists.txt**

Adapted from the official whisper.cpp Android example with paths corrected for our submodule placement at `app/src/main/cpp/whisper.cpp`:

```cmake
cmake_minimum_required(VERSION 3.22)
project(whisper_transchat)
set(CMAKE_CXX_STANDARD 17)

# whisper.cpp submodule is at the same level as this CMakeLists.txt
set(WHISPER_CPP_DIR ${CMAKE_CURRENT_SOURCE_DIR}/whisper.cpp)

set(SOURCE_FILES
    ${WHISPER_CPP_DIR}/src/whisper.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/whisper_jni.cpp
)

find_library(LOG_LIB log)
include(FetchContent)

function(build_library target_name)
    add_library(${target_name} SHARED ${SOURCE_FILES})
    target_compile_definitions(${target_name} PUBLIC GGML_USE_CPU)

    if (${target_name} STREQUAL "whisper_v8fp16_va")
        target_compile_options(${target_name} PRIVATE -march=armv8.2-a+fp16)
        set(GGML_COMPILE_OPTIONS -march=armv8.2-a+fp16)
    endif ()

    if (NOT ${CMAKE_BUILD_TYPE} STREQUAL "Debug")
        target_compile_options(${target_name} PRIVATE
            -O3
            -fvisibility=hidden
            -fvisibility-inlines-hidden
            -ffunction-sections
            -fdata-sections
        )
        target_link_options(${target_name} PRIVATE
            -Wl,--gc-sections
            -Wl,--exclude-libs,ALL
            -flto
        )
    endif ()

    # Required for Play Store and Pixel 9+ (16KB page size)
    target_link_options(${target_name} PRIVATE "-Wl,-z,max-page-size=16384")

    FetchContent_Declare(ggml SOURCE_DIR ${WHISPER_CPP_DIR}/ggml)
    FetchContent_MakeAvailable(ggml)
    target_compile_options(ggml PRIVATE ${GGML_COMPILE_OPTIONS})
    target_link_libraries(${target_name} ${LOG_LIB} android ggml)

    target_include_directories(${target_name} PRIVATE
        ${WHISPER_CPP_DIR}
        ${WHISPER_CPP_DIR}/src
        ${WHISPER_CPP_DIR}/include
        ${WHISPER_CPP_DIR}/ggml/include
        ${WHISPER_CPP_DIR}/ggml/src
        ${WHISPER_CPP_DIR}/ggml/src/ggml-cpu  # required since PR #2624 (Dec 2024)
    )
endfunction()

# arm64-v8a optimized variant (FP16 acceleration) — primary build target
if (${ANDROID_ABI} STREQUAL "arm64-v8a")
    build_library("whisper_v8fp16_va")
endif ()

# Baseline fallback target (always built)
build_library("whisper")
```

**Step 3: Create app/src/main/cpp/whisper_jni.cpp**

JNI function names use the package `com.simonpena.transchat` with inner class `WhisperLib$Companion` (`00024` = dollar sign encoding):

```cpp
#include <jni.h>
#include <string>
#include <android/log.h>
#include "whisper.cpp/include/whisper.h"

#define LOG_TAG "WhisperJNI"
#define LOGI(...) __android_log_print(ANDROID_LOG_INFO, LOG_TAG, __VA_ARGS__)
#define LOGE(...) __android_log_print(ANDROID_LOG_ERROR, LOG_TAG, __VA_ARGS__)

extern "C" {

// Initialize from absolute file path — always use this, never byte array (OOM on 190MB)
JNIEXPORT jlong JNICALL
Java_com_simonpena_transchat_WhisperLib_00024Companion_initContext(
        JNIEnv *env, jobject /* thiz */, jstring modelPathStr) {
    const char *modelPath = env->GetStringUTFChars(modelPathStr, nullptr);
    LOGI("Loading model from: %s", modelPath);
    struct whisper_context_params params = whisper_context_default_params();
    struct whisper_context *ctx = whisper_init_from_file_with_params(modelPath, params);
    env->ReleaseStringUTFChars(modelPathStr, modelPath);
    if (ctx == nullptr) {
        LOGE("Failed to initialize whisper context from file");
        return 0L;
    }
    LOGI("Whisper context initialized successfully");
    return (jlong)(intptr_t)ctx;
}

// Run transcription — blocking, call from background thread only (15-60s on mid-range)
JNIEXPORT void JNICALL
Java_com_simonpena_transchat_WhisperLib_00024Companion_fullTranscribe(
        JNIEnv *env, jobject /* thiz */, jlong contextPtr, jint numThreads, jfloatArray audioData) {
    auto *ctx = (struct whisper_context *)(intptr_t)contextPtr;
    jfloat *audioDataArr = env->GetFloatArrayElements(audioData, nullptr);
    jsize audioDataLength = env->GetArrayLength(audioData);

    struct whisper_full_params params = whisper_full_default_params(WHISPER_SAMPLING_GREEDY);
    params.n_threads        = numThreads;
    params.print_progress   = false;
    params.print_timestamps = false;
    params.no_context       = true;          // each clip is independent
    params.no_timestamps    = true;          // prevents hallucination amplification (Issue #1724)
    params.no_speech_thold  = 0.6f;         // discard near-silent segments
    params.max_tokens       = 224;           // cap repetition loops
    params.language         = nullptr;       // auto-detect (multilingual)
    params.detect_language  = true;
    params.temperature      = 0.0f;         // greedy decoding
    params.temperature_inc  = 0.2f;         // fallback increment

    LOGI("Starting transcription with %d threads on %d samples", numThreads, (int)audioDataLength);
    whisper_full(ctx, params, audioDataArr, (int)audioDataLength);
    env->ReleaseFloatArrayElements(audioData, audioDataArr, JNI_ABORT);
    LOGI("Transcription complete, segments: %d", whisper_full_n_segments(ctx));
}

JNIEXPORT jint JNICALL
Java_com_simonpena_transchat_WhisperLib_00024Companion_getTextSegmentCount(
        JNIEnv *env, jobject /* thiz */, jlong contextPtr) {
    auto *ctx = (struct whisper_context *)(intptr_t)contextPtr;
    return whisper_full_n_segments(ctx);
}

JNIEXPORT jstring JNICALL
Java_com_simonpena_transchat_WhisperLib_00024Companion_getTextSegment(
        JNIEnv *env, jobject /* thiz */, jlong contextPtr, jint index) {
    auto *ctx = (struct whisper_context *)(intptr_t)contextPtr;
    const char *text = whisper_full_get_segment_text(ctx, index);
    return env->NewStringUTF(text);
}

JNIEXPORT void JNICALL
Java_com_simonpena_transchat_WhisperLib_00024Companion_freeContext(
        JNIEnv *env, jobject /* thiz */, jlong contextPtr) {
    auto *ctx = (struct whisper_context *)(intptr_t)contextPtr;
    LOGI("Freeing whisper context");
    whisper_free(ctx);
}

JNIEXPORT jstring JNICALL
Java_com_simonpena_transchat_WhisperLib_00024Companion_getSystemInfo(
        JNIEnv *env, jobject /* thiz */) {
    return env->NewStringUTF(whisper_print_system_info());
}

} // extern "C"
```

**Step 4: Create app/proguard-rules.pro**

```
# Keep all JNI-facing native methods — R8 must not rename or strip these
-keepclassmembers class * {
    native <methods>;
}
-keepclasseswithmembernames class * {
    native <methods>;
}

# Keep the entire transchat package to preserve JNI name mangling
-keep class com.simonpena.transchat.** { *; }

# Keep WhisperLib companion object specifically
-keep class com.simonpena.transchat.WhisperLib { *; }
-keep class com.simonpena.transchat.WhisperLib$Companion { *; }
```
