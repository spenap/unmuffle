```kotlin
package com.simonpena.transchat

import android.content.res.AssetManager
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.asCoroutineDispatcher
import kotlinx.coroutines.withContext
import java.util.concurrent.Executors

/**
 * Low-level JNI bridge matching function signatures in whisper_jni.cpp.
 * Package name (com.simonpena.transchat) and inner class name (WhisperLib$Companion)
 * must stay in sync with the C++ JNI name mangling. Do NOT rename this class or its
 * companion — that would break the JNI binding without a compile error.
 *
 * CPU feature detection at init time loads the best available .so:
 *   arm64 with FP16 hardware → libwhisper_v8fp16_va.so
 *   everything else           → libwhisper.so (baseline)
 */
internal class WhisperLib {
    companion object {
        init {
            // Check for FP16 capability on arm64 to load the optimized variant
            // Reading /proc/cpuinfo is the standard Android approach (used in official example)
            val cpuInfo = try {
                java.io.File("/proc/cpuinfo").readText()
            } catch (e: Exception) {
                ""
            }
            val hasFp16 = cpuInfo.contains("fphp") || cpuInfo.contains("asimdhp")

            if (hasFp16) {
                try {
                    System.loadLibrary("whisper_v8fp16_va")
                } catch (e: UnsatisfiedLinkError) {
                    System.loadLibrary("whisper")  // fallback
                }
            } else {
                System.loadLibrary("whisper")
            }
        }

        // Initialize context from absolute file path — always prefer this over byte arrays (OOM risk)
        external fun initContext(modelPath: String): Long

        // Run full transcription — blocking, must be called from a non-main thread
        external fun fullTranscribe(contextPtr: Long, numThreads: Int, audioData: FloatArray)

        external fun getTextSegmentCount(contextPtr: Long): Int
        external fun getTextSegment(contextPtr: Long, index: Int): String
        external fun freeContext(contextPtr: Long)
        external fun getSystemInfo(): String
    }
}

/**
 * Wraps a native whisper_context pointer. Single-threaded executor prevents concurrent
 * access — whisper.cpp is NOT thread-safe and crashes on concurrent calls.
 *
 * Use WhisperEngine (Plan 02) rather than this class directly.
 */
internal class WhisperContext private constructor(private var ptr: Long) {
    // Dedicated single-thread executor — whisper.cpp context must not be accessed concurrently
    private val scope = CoroutineScope(
        Executors.newSingleThreadExecutor().asCoroutineDispatcher()
    )

    /**
     * Transcribes [data] (16kHz mono float32 PCM) and returns concatenated segment text.
     * This function suspends on the single-thread executor — it does NOT run on the caller's thread.
     */
    suspend fun transcribeData(data: FloatArray): String = withContext(scope.coroutineContext) {
        WhisperLib.fullTranscribe(ptr, WhisperCpuConfig.preferredThreadCount, data)
        val segmentCount = WhisperLib.getTextSegmentCount(ptr)
        buildString {
            for (i in 0 until segmentCount) {
                append(WhisperLib.getTextSegment(ptr, i))
            }
        }.trim()
    }

    /** Release native memory. Call when the engine is no longer needed. */
    suspend fun release() = withContext(scope.coroutineContext) {
        WhisperLib.freeContext(ptr)
        ptr = 0L
    }

    companion object {
        /**
         * Load model from an absolute file path. Returns null if the file does not exist
         * or whisper_init_from_file fails (corrupt model, out of memory, etc.).
         * Never pass a byte array — 190MB byte array causes OOM on all device tiers.
         */
        fun createContextFromFile(filePath: String): WhisperContext? {
            val ptr = WhisperLib.initContext(filePath)
            return if (ptr != 0L) WhisperContext(ptr) else null
        }

        fun getSystemInfo(): String = WhisperLib.getSystemInfo()
    }
}
```

```kotlin
package com.simonpena.transchat

/**
 * CPU thread count configuration for Whisper inference.
 *
 * Strategy: leave one core free for the UI/render thread, cap at 4.
 * Inference is memory-bandwidth-bound above 4 threads on most ARM SoCs;
 * more threads increase thermal pressure without improving throughput.
 */
internal object WhisperCpuConfig {
    val preferredThreadCount: Int
        get() = maxOf(1, minOf(Runtime.getRuntime().availableProcessors() - 1, 4))
}
```

```kotlin
package com.simonpena.transchat

import android.util.Log
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext

private const val TAG = "WhisperEngine"

/**
 * Public API for Whisper inference. Used by TranscriptionViewModel (Phase 3) and
 * the smoke test Activity (Plan 04).
 *
 * Threading contract:
 *   - loadModel() must be called from a coroutine (it does IO and JNI work)
 *   - transcribe() runs on Dispatchers.Default — never on Main (ANR after 5s)
 *   - release() frees native memory; call when Activity/ViewModel is destroyed
 *
 * Lifecycle: one engine per app session. The 190MB model stays resident in memory
 * once loaded — reloading on every transcription call would take 2-5s per call.
 */
class WhisperEngine {

    private var whisperContext: WhisperContext? = null

    /** True if the model has been loaded and the native context is ready. */
    val isModelLoaded: Boolean
        get() = whisperContext != null

    /**
     * Load the Whisper model from [absoluteFilePath].
     *
     * @param absoluteFilePath Must be an absolute path inside filesDir — e.g.
     *   `context.filesDir.absolutePath + "/models/ggml-small-q5_1.bin"`.
     *   Never pass a relative path; JNI will fail silently.
     * @return true on success, false if file missing or model load failed (OOM, corrupt)
     */
    suspend fun loadModel(absoluteFilePath: String): Boolean = withContext(Dispatchers.Default) {
        // Release any previously loaded context first
        whisperContext?.release()
        whisperContext = null

        Log.i(TAG, "Loading model: $absoluteFilePath")
        val ctx = try {
            WhisperContext.createContextFromFile(absoluteFilePath)
        } catch (e: Exception) {
            Log.e(TAG, "Exception during model load: ${e.message}", e)
            null
        }

        if (ctx == null) {
            Log.e(TAG, "Model load returned null — file missing, corrupt, or OOM")
            return@withContext false
        }

        whisperContext = ctx
        Log.i(TAG, "Model loaded successfully. System info: ${WhisperContext.getSystemInfo()}")
        true
    }

    /**
     * Transcribe [audioData] (16kHz mono float32 PCM) and return the transcript string.
     *
     * @throws IllegalStateException if [loadModel] has not been called successfully
     * @return Transcript text, or empty string if no speech detected (no_speech_thold applied)
     */
    suspend fun transcribe(audioData: FloatArray): String {
        val ctx = whisperContext
            ?: throw IllegalStateException("Model not loaded — call loadModel() first")

        val startMs = System.currentTimeMillis()
        Log.i(TAG, "Starting transcription of ${audioData.size} samples (${audioData.size / 16000.0f}s)")

        return withContext(Dispatchers.Default) {
            val result = ctx.transcribeData(audioData)
            val durationMs = System.currentTimeMillis() - startMs
            Log.i(TAG, "Transcription complete in ${durationMs}ms, result: '${result.take(80)}'")
            result
        }
    }

    /**
     * Release native Whisper context and free ~190MB of native memory.
     * Call from ViewModel.onCleared() or Application.onTerminate().
     */
    suspend fun release() {
        whisperContext?.release()
        whisperContext = null
        Log.i(TAG, "WhisperEngine released")
    }
}
```
