```kotlin
    /**
     * Transcribes [data] (16kHz mono float32 PCM) and returns concatenated segment text.
     * This function suspends on the single-thread executor — it does NOT run on the caller's thread.
     *
     * @param onProgress Called with 0-100 progress values during inference. Invoked on
     *   Dispatchers.Default, NOT the main thread. Default no-op for backward compatibility.
     */
    suspend fun transcribeData(
        data: FloatArray,
        onProgress: (Int) -> Unit = {}
    ): String = withContext(scope.coroutineContext) {
        // Poller runs on Default, NOT the single-thread executor (which is blocked by fullTranscribe)
        val pollerJob = launch(Dispatchers.Default) {
            while (isActive) {
                val p = WhisperLib.getProgress()
                onProgress(p)
                delay(200)
            }
        }
        try {
            WhisperLib.fullTranscribe(ptr, WhisperCpuConfig.preferredThreadCount, data)
        } finally {
            pollerJob.cancel()
        }
        onProgress(100) // Ensure 100% is reported before collecting segments
        val segmentCount = WhisperLib.getTextSegmentCount(ptr)
        buildString {
            for (i in 0 until segmentCount) {
                append(WhisperLib.getTextSegment(ptr, i))
            }
        }.trim()
    }
```

4. Modify `WhisperEngine.transcribe()` to accept an `onProgress` parameter. Replace the existing method:

```kotlin
    /**
     * Transcribe [audioData] (16kHz mono float32 PCM) and return the transcript string.
     *
     * @param onProgress Called with 0-100 progress values during inference. Default no-op.
     * @throws IllegalStateException if [loadModel] has not been called successfully
     * @return Transcript text, or empty string if no speech detected (no_speech_thold applied)
     */
    suspend fun transcribe(audioData: FloatArray, onProgress: (Int) -> Unit = {}): String {
        val ctx = whisperContext
            ?: throw IllegalStateException("Model not loaded — call loadModel() first")

        val startMs = System.currentTimeMillis()
        Log.i(TAG, "Starting transcription of ${audioData.size} samples (${audioData.size / 16000.0f}s)")

        return withContext(Dispatchers.Default) {
            val result = ctx.transcribeData(audioData, onProgress)
            val durationMs = System.currentTimeMillis() - startMs
            Log.i(TAG, "Transcription complete in ${durationMs}ms, result: '${result.take(80)}'")
            result
        }
    }
```

The only change to WhisperEngine.transcribe() is the added `onProgress: (Int) -> Unit = {}` parameter and forwarding it to `ctx.transcribeData(audioData, onProgress)`. The default value `= {}` ensures backward compatibility — existing callers (if any beyond the ViewModel) do not need to change.
